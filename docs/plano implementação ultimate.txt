class DataMigrationService {
  static Future<void> migrateFromSharedPreferences() async {
    final prefs = await SharedPreferences.getInstance();
    final moduleBox = Hive.box<UserModuleStatus>('module_states');
    
    // Migrar dados existentes para cada módulo real
    for (final nicheId in NicheId.values) {
      final key = '${nicheId.name}_status';
      final json = prefs.getString(key);
      if (json != null) {
        final status = UserModuleStatus.fromJson(jsonDecode(json));
        await moduleBox.put(nicheId.name, status);
      }
    }
  }
}

class MemoryMonitor {

      final info = ProcessInfo.currentRss;
      LoggerService.instance.performance('Memory usage: ${info ~/ 1024 / 1024} MB');

class InsigniaService {
  Future<void> checkAndUnlockInsignia(NicheId nicheId, int streakDays) async {
    FocusInsignia? newInsignia;
    
    if (streakDays >= 90) {
      newInsignia = FocusInsignia.disciplinum;
    } else if (streakDays >= 30) {
      newInsignia = FocusInsignia.diamante;
    } else if (streakDays >= 21) {
      newInsignia = FocusInsignia.ouro;
    } else if (streakDays >= 14) {
      newInsignia = FocusInsignia.prata;
    } else if (streakDays >= 7) {
      newInsignia = FocusInsignia.bronze;
    } else if (streakDays >= 5) {
      newInsignia = FocusInsignia.latao;
    } else if (streakDays >= 3) {
      newInsignia = FocusInsignia.aluminio;
    } else if (streakDays >= 1) {
      newInsignia = FocusInsignia.ferro;
    }
    
    if (newInsignia != null) {
      await _unlockInsignia(nicheId, newInsignia);
    }
  }
}

CountdownService countdownService(CountdownServiceRef ref) {

class CountdownService {
  Future<void> startCountdown(String packageName, NicheId nicheId, int duration) async {
    final session = DetectionSession(
      packageName: packageName,
      nicheId: nicheId,
      startTime: DateTime.now(),
      duration: duration,
      isActive: true,
    );
    
    await _timerRepository.saveSession(session);
    await _startTimer(session);
  }
  
  Future<void> recoverActiveTimers() async {
    final activeSessions = await _timerRepository.getActiveSessions();
    
    for (final session in activeSessions) {
      final elapsed = DateTime.now().difference(session.startTime);
      final remaining = session.duration - elapsed.inSeconds;
      
      if (remaining > 0) {
        await _startTimer(session, remaining);
      } else {
        await _onTimerExpired(session);
      }
    }
  }
}

1️⃣ Risco de crash silencioso no EventChannel (Accessibility)
Onde aparece

No seu serviço Android:

private var eventSink: EventChannel.EventSink? = null

Eventos são enviados assim:

eventSink?.success(packageName)
Problema

Se o Flutter reiniciar a engine ou o canal for recriado:

eventSink fica referenciando um canal morto

eventos são perdidos

ou ocorre crash silencioso

Isso é comum quando:

app vai para background

Flutter engine reinicia

Activity recria

Solução profissional

Nunca manter EventSink como variável global estática.

Use StreamHandler lifecycle correto.

Exemplo seguro:

class AccessibilityStreamHandler : EventChannel.StreamHandler {

    private var sink: EventChannel.EventSink? = null

    override fun onListen(arguments: Any?, events: EventChannel.EventSink?) {
        sink = events
    }

    override fun onCancel(arguments: Any?) {
        sink = null
    }

    fun sendPackage(packageName: String) {
        sink?.success(packageName)
    }
}

Isso evita:

memory leak

eventos enviados para canal morto

  String name;

Cada módulo implementa isso.

PornBlockModule implements DisciplinumModule
